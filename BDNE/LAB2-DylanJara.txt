//g. Crear una consulta con agregación usando match, project y sort con enunciado
//Queremos ver las bencineras Shell con los precios de 95 más bajo en las regiones que contengan la cadena "AR"
db.CNE.aggregate([{$match:{"distribuidor.nombre":/shell/i,"nombre_region":/ar/i,"precios.gasolina 95":{$gt:0}}},
  {$project:{
    _id:0, "distribuidor.nombre":1, nombre_region:1, "precios.gasolina 95":1}},{$sort:{"precios.gasolina 95":1}}  
])

//h. Crear una consulta con agregación usando group a dos claves, project y sort con enunciado
//Buscamos la gasolinera de cada region que contan el substring AR que venda la bencina de 97 más barata y ordenamos de menor a mayor
db.CNE.aggregate([{$match:{"nombre_region":/ar/i, "precios.gasolina 97":{$gt:0}}},
  {$sort:{"precios.gasolina 97":1}},
  {$group: 
    {_id:"$nombre_region", Distribuidor:{$first:"$distribuidor.nombre"}, Precio: {$first:"$precios.gasolina 97"}}},
  {$project:{
    _id:0, "nombre_region":"$_id", "distribuidor.nombre":"$Distribuidor", "precios.gasolina 97":"$Precio"}},
  {$sort:{"precios.gasolina 97":1}}  
])

//i. Crear una consulta con agregación usando group y promedio con enunciado
//Buscamos el valor promedio de la bencina de 93 por comuna en todo Chile
db.CNE.aggregate([{$group:{_id:"$nombre_comuna", Promedio:{$avg:"$precios.gasolina 93"}}},
{$project: {_id: 0, Comuna:"$_id", Promedio:1}}
])
